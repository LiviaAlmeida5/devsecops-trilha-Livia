name: Continuos Integration

on:
  push:
  pull_request:

permissions:
  contents: read

jobs:
  build-and-sast:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          persist-credentials: false

      - name: Usar Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: npm
          cache-dependency-path: app/package-lock.json

      - name: Instalar dependências
        working-directory: app
        run: npm ci

      - name: SAST - bloquear achados WARNING e ERROR
        run: |
          docker run --rm \
            -v "$GITHUB_WORKSPACE:/src" \
            -w /src \
            semgrep/semgrep:latest \
            semgrep scan \
              --config p/javascript \
              --config p/security-audit \
              --severity WARNING \
              --severity ERROR \
              --error \
              --strict \
              --metrics=off \
              app/src
